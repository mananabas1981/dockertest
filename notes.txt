CREATE TABLE "remittance_base" (
    "id"    TEXT,
    "code"  TEXT,
    "name"  TEXT,
    "from_" TEXT,
    "to_"   TEXT,
    "refids"    JSON,
    "createdby" TEXT,
    "datecreated"   TEXT,
    "status"    TEXT,
    PRIMARY KEY("id")
);

CREATE TABLE "remittance_item" (
    "remittance_id" TEXT,
    "emp_id"    TEXT,
    "data"  JSON,
    "addedby"   TEXT,
    "createdby" TEXT,
    "status"    TEXT,
    PRIMARY KEY("remittance_id","emp_id")
);
